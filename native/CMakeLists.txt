cmake_minimum_required(VERSION 3.22.1)
project(codebrix_android_audio LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(CODEBRIX_CODECS_ONLY "Build codecs on the host for device-free tests" OFF)
add_library(codebrix_miniaudio SHARED codecs.c)
target_link_libraries(codebrix_miniaudio PRIVATE m)
if(NOT CODEBRIX_CODECS_ONLY)
  if(NOT ANDROID)
    message(FATAL_ERROR "The Oboe backend requires the Android NDK toolchain")
  endif()
  set(BUILD_SHARED_LIBS OFF CACHE BOOL "Build Oboe statically" FORCE)
  add_subdirectory(vendor/oboe)
  target_sources(codebrix_miniaudio PRIVATE bridge.cpp)
  target_link_libraries(codebrix_miniaudio PRIVATE oboe log android)
  target_link_options(codebrix_miniaudio PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,-z,common-page-size=16384")
endif()
